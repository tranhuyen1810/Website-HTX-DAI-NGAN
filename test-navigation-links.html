CSS Dropdown Show:

.factory-menu .has-dropdown:hover > .dropdown-menu {
    opacity: 1;
    visibility: visible;
    transform: translateY(0);
}

/* Không có pointer-events: none trên dropdown */

JavaScript Active State:

// Tự động detect trang hiện tại
const currentPage = window.location.pathname.split('/').pop();

// Tìm link tương ứng và đánh dấu active
document.querySelectorAll('.factory-menu a').forEach(link => {
    if (link.getAttribute('href') === currentPage) {
        link.closest('li').classList.add('active');
    }
});
